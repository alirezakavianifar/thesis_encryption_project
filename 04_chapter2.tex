\chapter{مفاهیم پایه و پیشینه تحقیق}

\section{مقدمه}
در این فصل، به بررسی مفاهیم بنیادی رمزنگاری، ویژگی‌های تصاویر دیجیتال و مبانی نظریه‌ی آشوب پرداخته می‌شود. همچنین، ارتباط میان سیستم‌های آشوبی و الگوریتم‌های رمزنگاری تشریح شده و در نهایت، پیشینه‌ای از پژوهش‌های انجام‌شده در این حوزه ارائه می‌گردد.

\section{رمزنگاری}
رمزنگاری (\lr{Cryptography}) دانشی است که به مطالعه و توسعه‌ی روش‌هایی برای برقراری ارتباط امن در حضور اشخاص ثالث (مهاجمان) می‌پردازد. هدف اصلی رمزنگاری، تأمین محرمانگی (\lr{Confidentiality})، یکپارچگی (\lr{Integrity})، و احراز اصالت (\lr{Authentication}) داده‌ها است.

\section{رمزنگاری تصویر دیجیتال}
با گسترش اینترنت و شبکه‌های اجتماعی، تبادل تصاویر دیجیتال افزایش یافته است. تصاویر دیجیتال دارای ویژگی‌هایی هستند که آن‌ها را از داده‌های متنی متمایز می‌کند.

\subsection{محدودیت‌های روش‌های سنتی برای تصاویر}
الگوریتم‌های کلاسیک نظیر \lr{RSA} و \lr{AES} برای رمزنگاری متن طراحی شده‌اند. استفاده مستقیم از این روش‌ها برای تصاویر به دلایل زیر چالش‌برانگیز است:
\begin{itemize}
    \item \textbf{حجم بالای داده:} تصاویر دیجیتال معمولاً شامل تعداد زیادی پیکسل هستند که پردازش آن‌ها با روش‌های سنتی زمان‌بر است.
    \item \textbf{همبستگی بالای پیکسل‌ها:} در یک تصویر طبیعی، پیکسل‌های مجاور همبستگی شدیدی دارند که می‌تواند منجر به نشت اطلاعات در صورت استفاده از رمزنگاری ضعیف شود.
    \item \textbf{یکنواختی رنگ:} بخش‌های بزرگی از تصاویر ممکن است رنگ یکسانی داشته باشند که در هیستوگرام تصویر رمزشده بازتاب می‌یابد.
\end{itemize}

\subsection{خصوصیات تصویر دیجیتال}
هر تصویر دیجیتال از کوچک‌ترین واحدهایی به نام پیکسل (\lr{Pixel}) تشکیل شده است. 
\begin{itemize}
    \item \textbf{عمق رنگ:} تعداد بیت‌های اختصاص یافته به هر پیکسل برای نمایش رنگ (مثلاً ۸ بیت برای تصاویر خاکستری و ۲۴ بیت برای تصاویر رنگی \lr{RGB}).
    \item \textbf{ابعاد تصویر:} تعداد پیکسل‌ها در طول و عرض تصویر (مثلاً $512 \times 512$).
    \item \textbf{مدل‌های رنگ:} رایج‌ترین مدل برای پردازش تصویر، مدل \lr{RGB} است که شامل سه کانال قرمز، سبز و آبی می‌باشد.
\end{itemize}

\section{نظریه‌ی آشوب و مفاهیم بنیادی آن}
نظریه‌ی آشوب (\lr{Chaos Theory}) به مطالعه سیستم‌های دینامیکی غیرخطی می‌پردازد که علی‌رغم ماهیت قطعی (\lr{Deterministic})، رفتاری بسیار پیچیده و غیرقابل‌پیش‌بینی از خود نشان می‌دهند.

\subsection{ویژگی‌های سیستم‌های آشوبی}
سیستم‌های آشوبی دارای ویژگی‌های منحصربه‌فردی هستند:
\begin{enumerate}
    \item \textbf{حساسیت شدید به شرایط اولیه:} این ویژگی که به «اثر پروانه‌ای» (\lr{Butterfly Effect}) معروف است، بیان می‌کند که تغییر بسیار کوچک در مقادیر اولیه منجر به تفاوت‌های فاحش در خروجی سیستم در بلندمدت می‌شود.
    \item \textbf{اغتشاش (\lr{Ergodicity}):} مسیرهای سیستم در فضای فاز به‌گونه‌ای توزیع می‌شوند که تمام فضا را پوشش می‌دهند.
    \item \textbf{شبه‌تصادفی بودن:} خروجی سیستم‌های آشوبی از نظر آماری بسیار شبیه به نویز تصادفی است، در حالی که توسط معادلات معین تولید شده‌اند.
\end{enumerate}

\section{ارتباط بین آشوب و رمزنگاری}
شباهت‌های زیادی میان سیستم‌های آشوبی و سیستم‌های رمزنگاری وجود دارد که در جدول زیر خلاصه شده است:

\begin{table}[h]
\centering
\caption{شباهت‌های سیستم‌های آشوبی و رمزنگاری}
\begin{tabular}{|c|c|}
\hline
\textbf{سیستم‌های آشوبی} & \textbf{سیستم‌های رمزنگاری} \\ \hline
اغتشاش (\lr{Ergodicity}) & سردرگمی (\lr{Confusion}) \\ \hline
حساسیت به شرایط اولیه & انتشار (\lr{Diffusion}) \\ \hline

\end{table}

\section{پیشینه پژوهش}
در سال‌های اخیر، پژوهش‌های متعددی در زمینه رمزنگاری تصویر بر پایه آشوب انجام شده است. 
\begin{itemize}
    \item \textbf{ژانگ و لیو [3]:} ترکیب سیستم‌های آشوبی \lr{Henon} و \lr{Tent} را برای افزایش امنیت پیشنهاد دادند.
    \item \textbf{هو و ژو [4]:} مدل آشوبی نمایی (\lr{ECM}) را برای تولید نگاشت‌های آشوبناک پایدار معرفی کردند که مبنای الگوریتم پیشنهادی در این پژوهش است.
    \item \textbf{خدادادی و زندوکیلی [7]:} روشی ترکیبی از سیستم‌های آشوبی برای رمزنگاری تصاویر رنگی ارائه دادند که در آن از سیستم \lr{Chen} برای جایگشت استفاده شده است.
\end{itemize}
