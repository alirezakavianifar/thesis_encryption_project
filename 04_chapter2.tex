\chapter{مفاهیم پایه و پیشینه تحقیق}

\section{مقدمه}
از زمان انتشار مشاهده‌های ادوارد لورنز در زمینه‌ی رفتارهای آشوبناک، مفهوم «آشوب» به یکی از موضوعات مورد توجه در جامعه‌ی علمی تبدیل شد. نظریه‌ی آشوب شاخه‌ای از ریاضیات است که به تحلیل سیستم‌هایی می‌پردازد که رفتار آن‌ها به تغییرات کوچک در شرایط اولیه بسیار حساس است. پدیده‌هایی که در مقیاس کوچک آشفته به نظر می‌رسند، ممکن است در مقیاس بزرگ‌تر رفتاری منظم و پایدار داشته باشند.

\section{رمزنگاری}
رمزنگاری (\lr{Cryptography}) از دو واژه‌ی یونانی \lr{krypto} به معنی «پنهان» و \lr{grapho} به معنی «نوشتن» تشکیل شده است.

\subsection{تعریف رمزنگاری}
رمزنگاری فرآیندی است که طی آن اطلاعات به‌گونه‌ای دچار تغییر و بازآرایی می‌شوند که برای افراد غیرمجاز غیرقابل فهم باشد. هدف اصلی رمزنگاری، طراحی طرح‌ها و پروتکل‌هایی است که حتی در حضور دشمن نیز امکان انجام فعالیت‌های خاص را فراهم سازد.

\subsection{شیوه‌های پایه رمزنگاری}
در رمزنگاری کلاسیک، دو شیوه‌ی پایه وجود دارد:
\begin{enumerate}
    \item \textbf{رمزنگاری جابجایی (\lr{Transposition}):} در این روش متن اصلی بدون تغییر محتوا، به‌صورت ترتیبی جابجا می‌شود.
    \item \textbf{رمزنگاری جایگزینی (\lr{Substitution}):} هر واحد از متن اصلی با یک نماد دیگر طبق یک سیستم مشخص جایگزین می‌شود.
\end{enumerate}

\subsection{انواع روش‌های رمزنگاری بر اساس نوع کلید}
الگوریتم‌های رمزنگاری به دو دسته متقاران (\lr{Symmetric}) و نامتقاران (\lr{Asymmetric}) تقسیم می‌شوند.

\begin{table}[h]
\centering
\caption{مقایسه رمزنگاری متقارن و نامتقارن}
\begin{tabular}{|l|l|l|}
\hline
\textbf{ویژگی} & \textbf{رمزنگاری متقارن} & \textbf{رمزنگاری نامتقارن} \\ \hline
نوع کلید & یک کلید & یک کلید عمومی و یک کلید خصوصی \\ \hline
نیاز به کانال امن & نیازمند کانال امن & نیاز به کانال امن ندارد \\ \hline

\end{tabular}

\subsection{ویژگی‌های یک الگوریتم رمزنگاری امن}
یک الگوریتم امن باید دارای ویژگی‌های زیر باشد:
\begin{enumerate}
    \item \textbf{یک‌طرفه بودن:} دستیابی به متن اصلی بدون کلید غیرممکن باشد.
    \item \textbf{عدم وابستگی امنیت به الگوریتم:} طبق اصل کرشهف، امنیت نباید به مخفی بودن الگوریتم وابسته باشد.
    \item \textbf{وابستگی به تمامی بیت‌ها:} کوچک‌ترین تغییر در متن اصلی یا کلید باید خروجی کاملاً متفاوتی تولید کند.
    \item \textbf{ساختار جبری پیچیده:} برای جلوگیری از تحلیل‌های ریاضی ساده.
    \item \textbf{برابری طول متن:} طول پیام ورودی و خروجی باید حفظ شود.
\end{enumerate}

\subsection{تحلیل رمز: انواع حملات}
حملات به چهار دسته اصلی تقسیم می‌شوند:
\begin{enumerate}
    \item \textbf{حمله با متن رمزشده (\lr{Ciphertext-Only}):} تحلیلگر تنها به متن رمزشده دسترسی دارد.
    \item \textbf{حمله با متن اصلی مشخص (\lr{Known-Plaintext}):} دسترسی به جفت‌های متن اصلی و رمز.
    \item \textbf{حمله با متن اصلی منتخب (\lr{Chosen-Plaintext}):} تحلیلگر متن اصلی را برای رمزگذاری انتخاب می‌کند.
    \item \textbf{حمله با متن رمزشده منتخب (\lr{Chosen-Ciphertext}):} حالت تطبیقی حمله قبلی.
\end{enumerate}

\subsection{امنیت سیستم‌های رمزنگاری}
\subsubsection{اصول شش‌گانه کرشهف}
آگوست کرشهف در سال ۱۸۸۳ اصول زیر را مطرح کرد:
۱. غیرقابل شکست بودن سیستم. ۲. محرمانه بودن تنها کلید رمز. ۳. قابلیت انتخاب و تغییر کلید. ۴. امکان ارسال از طریق تلگراف. ۵. قابلیت حمل و نقل. ۶. سهولت راه‌اندازی.

\subsubsection{معیارهای پنجگانه شانون}
کلاد شانون در سال ۱۹۴۸ معیارهای زیر را معرفی نمود:
۱. میزان ایمنی سیستم. ۲. اندازه کلید. ۳. پیچیدگی عملیات. ۴. انتشار خطا. ۵. بسط یا گسترش پیام.

\section{رمزنگاری تصویر دیجیتال}
تصاویر دیجیتال به دلیل حجم بالا و همبستگی شدید پیکسل‌های مجاور، نیازمند الگوریتم‌های اختصاصی هستند.

\subsection{محدودیت‌های روش‌های سنتی}
روش‌هایی مانند \lr{AES} و \lr{RSA} برای تصاویر کارا نیستند به دلیل: ۱. حجم بالای تصاویر. ۲. زمان پردازش طولانی. ۳. همبستگی پیکسل‌های مجاور.

\subsection{خصوصیات تصویر دیجیتال}
تصاویر از پیکسل‌ها تشکیل شده‌اند. مفاهیمی مانند عمق رنگ (تعداد رنگ‌های قابل تولید) و ابعاد تصویر (تعداد پیکسل‌های طول و عرض) در امنیت آن‌ها نقش دارند. مدل رنگ \lr{RGB} (قرمز، سبز، آبی) پرکاربردترین مدل برای پردازش تصویر است.

\section{نظریه‌ی آشوب و مفاهیم بنیادی آن}
نظریه آشوب با پژوهش‌های ادوارد لورنز و کشف «اثر پروانه‌ای» به اوج رسید. این نظریه بیان می‌کند که در دل بی‌نظمی، نوعی نظم نهفته وجود دارد.

\subsection{ویژگی‌های تئوری آشوب}
\begin{itemize}
    \item \textbf{اثر پروانه‌ای:} حساسیت شدید به شرایط اولیه.
    \item \textbf{سازگاری پویا:} توانایی تطبیق با محیط.
    \item \textbf{جذب‌کننده عجیب:} ساختارهای پیچیده فراکتالی که مسیرهای سیستم به سمت آن‌ها میل می‌کنند.
    \item \textbf{خودتشابهی:} تکرار الگوها در مقیاس‌های مختلف.
\end{itemize}

\subsection{نمای لیاپانوف}
این شاخص میانگین نرخ واگرایی یا همگرایی دو مسیر بسیار نزدیک را نشان می‌دهد و معیار اصلی برای تشخیص وجود آشوب در یک سیستم است.

\subsection{نمونه‌هایی از سیستم‌های آشوبی}
نگاشت لجستیک (یک‌بعدی)، سیستم لورنز (سه‌بعدی برای مدل‌سازی جو) و سیستم \lr{Chen} (سه‌بعدی با پیچیدگی بالاتر) از نمونه‌های اصلی هستند.

\section{پیشینه پژوهش}
پژوهش‌های اخیر بر استفاده از نقشه‌های آشوبی نمایی (\lr{ECM}) متمرکز شده‌اند. ژانگ و لیو (۲۰۲۳) ترکیب \lr{Henon} و \lr{Tent} را پیشنهاد دادند. هو و ژو (۲۰۱۹) مدل \lr{ECM} را برای پایداری بیشتر معرفی کردند. خدادادی و زندوکیلی (۲۰۱۹) نیز روشی مبتنی بر دو سیستم \lr{Chen} ارائه کردند که کارایی بالایی در امنیت تصاویر رنگی نشان داده است.
