\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\setcounter{enumi}{1}
\item
  \protect\hypertarget{_Toc217957704}{}{}\RL{رمزنگاری}
\end{enumerate}

\RL{رمزنگاری یا رمزگذاری فرآیندی است که طی آن اطلاعات به‌کمک یک الگوریتم خاص به صورتی غیرقابل‌فهم برای افراد غیرمجاز تبدیل و ذخیره می‌شود تا از دسترسی یا سوء‌استفاده‌ی احتمالی از آن جلوگیری گردد.}

\subsection*{\RL{2-2-1- تعریف رمزنگاری}}

\RL{رمزنگاری فرآیندی است که طی آن اطلاعات به‌گونه‌ای دچار تغییر و بازآرایی می‌شوند که برای افراد غیرمجاز غیرقابل فهم باشد.}

\subsection*{\RL{2-2-2- شیوه‌های پایه رمزنگاری}}

\RL{در رمزنگاری کلاسیک، دو شیوه‌ی پایه وجود دارد: رمزنگاری جابجایی و رمزنگاری جایگزینی.}

\subsection*{\RL{2-2-3- انواع روش‌های رمزنگاری بر اساس نوع کلید}}

\RL{بر اساس نوع کلید، الگوریتم‌های رمزنگاری به دو دسته تقسیم می‌شوند: رمزنگاری متقارن و رمزنگاری نامتقارن.}
