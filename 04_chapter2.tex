\chapter{مفاهیم پایه و پیشینه تحقیق}

\section{مقدمه}
از زمان انتشار مشاهدات ادوارد لورنز در زمینه‌ی رفتارهای آشوبناک، مفهوم «آشوب» به یکی از موضوعات مورد توجه در جامعه‌ی علمی تبدیل شد. نظریه‌ی آشوب شاخه‌ای از ریاضیات است که به تحلیل سیستم‌هایی می‌پردازد که رفتار آن‌ها به تغییرات کوچک در شرایط اولیه بسیار حساس است. به بیان دیگر، در این‌گونه سامانه‌ها، تغییرات ظاهراً ناچیز می‌تواند منجر به تفاوت‌های چشمگیر در خروجی شود.

پیش از ظهور این نظریه، بسیاری از پدیده‌های تصادفی یا غیرقابل‌پیش‌بینی، فاقد توجیه علمی دقیق تلقی werden، اما نظریه‌ی آشوب توانست با اتکا بر اصول ریاضی، فیزیک و فلسفه، مبانی تبیینی منسجمی برای این پدیده‌ها فراهم آورد. قانون علت و معلول همچنان در این نظریه برقرار است، اما تعداد عوامل مؤثر بر نتایج به‌طور چشمگیری افزایش می‌یابد. در واقع، نظریه‌ی آشوب ابزاری فراهم می‌کند تا بتوان پدیده‌های پیچیده‌ی طبیعی و واقعی را بهتر درک و تحلیل کرد.

در سال‌های اخیر، توجه ویژه‌ای به کاربرد نظریه‌ی آشوب در علم ارتباطات و امنیت اطلاعات شده است. ویژگی‌هایی چون حساسیت بالا به شرایط اولیه، غیرقابل‌پیش‌بینی بودن و رفتار شبه‌تصادفی، سبب شده تا سیستم‌های آشوبی به‌عنوان بستری مناسب برای طراحی الگوریتم‌های رمزنگاری مطرح شوند. در این فصل به معرفی مبانی نظری مرتبط با موضوع پژوهش پرداخته می‌شود. ابتدا مفاهیم بنیادین رمزنگاری و انواع روش‌های آن مرور می‌گردد، سپس ویژگی‌های اصلی تصاویر دیجیتال، نظریه‌ی آشوب، انواع توابع آشوب و ارتباط میان آشوب و رمزنگاری مورد بررسی قرار می‌گیرد. در ادامه نیز به بررسی پژوهش‌های پیشین در زمینه‌ی استفاده از سیستم‌های آشوبی در رمزنگاری تصاویر رنگی پرداخته خواهد شد.

\section{رمزنگاری}
رمزنگاری (\lr{Cryptography}) از دو واژه‌ی یونانی \lr{krypto} به معنی «پنهان» و \lr{grapho} به معنی «نوشتن» تشکیل شده است. در مفهوم علمی، رمزنگاری به معنای تبدیل متن آشکار (\lr{Plain Text}) به متن رمز (\lr{Cipher Text}) و برعکس است.

\subsection{ویژگی‌های یک الگوریتم رمزنگاری امن}
در طراحی الگوریتم‌های رمزنگاری، رعایت ویژگی‌های زیر برای تضمین امنیت ضروری است:
\begin{enumerate}
    \item \textbf{یک‌طرفه بودن الگوریتم:} دستیابی به متن اصلی از طریق متن رمز بدون داشتن کلید، غیرممکن باشد.
    \item \textbf{عدم وابستگی امنیت به محرمانه بودن الگوریتم:} مخفی نگاه داشتن جزئیات الگوریتم در امنیت آن نقش ندارد و امنیت باید صرفاً بر اساس کلید مخفی تضمین شود.
    \item \textbf{وابستگی هر بیت متن رمزشده به تمامی بیت‌های متن اصلی:} کوچک‌ترین تغییر در متن اصلی باید متن رمزشده کاملاً متفاوتی تولید کند. الگوریتم‌هایی که این ویژگی را دارند، به عنوان \textbf{رمز کامل} (\lr{Perfect Cipher}) شناخته می‌شوند.
    \item \textbf{وابستگی هر بیت متن رمزشده به تمامی بیت‌های کلید سری:} کوچک‌ترین تغییر در کلید نیز باعث تولید متن رمزشده متفاوت می‌شود.
    \item \textbf{تأثیر تغییرات کوچک ورودی بر خروجی:} تغییر حتی یک بیت در داده‌های ورودی، بدون تغییر کلید، باید تغییرات عمده‌ای در داده‌های خروجی ایجاد کند.
    \item \textbf{تأثیر تغییرات کوچک کلید بر متن رمزشده:} تغییر حتی یک بیت در کلید سری، بدون تغییر متن اصلی، باید باعث ایجاد تغییرات عمده در متن رمزشده شود.
    \item \textbf{استفاده از عملیات جانشینی و جابجایی بیت‌ها تحت کنترل کلید و داده ورودی:} الگوریتم باید قادر باشد بیت‌ها را جایگزین و جابجا کند تا امنیت اطلاعات تضمین شود.
    \item \textbf{دارا بودن ساختار جبری پیچیده:} اگر تابع رمزنگاری دارای رابطه جبری ساده باشد، الگوریتم به راحتی قابل تحلیل و شکستن خواهد بود.
    \item \textbf{برابری طول متن اصلی و متن رمزشده:} الگوریتم باید طول پیام ورودی و خروجی را حفظ کند.
    \item \textbf{استفاده از تمامی کلیدهای مخفی برای تولید رمز قوی:} هر کلید باید توانایی تولید متن رمزشده قوی و غیرقابل پیش‌بینی را داشته باشد.
\end{enumerate}

\subsection{تحلیل رمز: انواع حملات به سامانه‌های رمزنگاری}
انواع حملات به چهار دسته اصلی تقسیم می‌شوند:
\begin{enumerate}
    \item \textbf{حمله با متن رمزشده (\lr{Ciphertext-Only Attack}):} در این نوع حمله، تحلیلگر تنها به متن رمزشده دسترسی دارد و تلاش می‌کند متن اصلی یا کلید مورد استفاده را بیابد. این حمله سخت‌ترین نوع حمله است و نیازمند تحلیل دقیق الگوها و احتمالات است.
    \item \textbf{حمله با متن اصلی مشخص (\lr{Known-Plaintext Attack}):} تحلیلگر به یک یا چند جفت متن اصلی و متن رمزشده متناظر دسترسی دارد و از آن‌ها برای حدس زدن کلید یا الگوریتم رمزنگاری استفاده می‌کند.
    \item \textbf{حمله با متن اصلی منتخب (\lr{Chosen-Plaintext Attack}):} تحلیلگر می‌تواند متن‌های اصلی خاصی را برای رمزگذاری انتخاب کند و متن‌های رمزشده متناظر را مشاهده نماید. این حمله از نوع \lr{Known-Plaintext} قوی‌تر است.
    \item \textbf{حمله با متن رمزشده منتخب (\lr{Chosen-Ciphertext Attack}):} تحلیلگر می‌تواند متن‌های اصلی را انتخاب و خروجی رمزگذاری را مشاهده کند و همچنین انتخاب‌های بعدی خود را بر اساس نتایج قبلی اصلاح نماید.
\end{enumerate}

\subsection{امنیت سیستم‌های رمزنگاری}
\subsubsection{اصول شش‌گانه کرشهف}
آگوست کرشهف در سال ۱۸۸۳ شش اصل اساسی را مطرح کرد:
\begin{enumerate}
    \item غیرقابل شکست بودن سیستم: سیستم رمزنگاری باید حتی در عمل غیرقابل شکست باشد.
    \item محرمانه بودن تنها کلید رمز: هیچ جزئیات دیگری از سیستم نباید مخفی باشد و تنها کلید، اطلاعات محرمانه است.
    \item قابلیت انتخاب و تغییر کلید: کلید باید به‌راحتی قابل تغییر و حفظ باشد.
    \item امکان ارسال از طریق تلگراف: متون رمزشده باید قابلیت مخابره از طریق رسانه‌های مشابه را داشته باشند.
    \item قابلیت حمل و نقل: دستگاه رمزنگاری یا اسناد رمز شده باید توسط یک نفر قابل حمل باشد.
    \item سهولت راه‌اندازی سیستم: سیستم باید به سادگی قابل استفاده باشد.
\end{enumerate}

در توضیح قانون دوم کرشهف، دلایل عمده پنهان کردن کلید به جای الگوریتم عبارتند از:
\begin{itemize}
    \item سادگی حفظ محرمانگی کلید نسبت به الگوریتم.
    \item تجدید امنیت در صورت افشای کلید (تنها با تعویض کلید).
    \item امکان استفاده از سیستم مشترک برای چندین گیرنده/فرستنده با کلیدهای متفاوت.
    \item امکان بازرسی و اصلاح الگوریتم‌های عمومی توسط محققان امنیتی برای یافتن حفره‌ها.
\end{itemize}

\subsubsection{معیارهای پنجگانه شانون}
کلاد شانون در سال ۱۹۴۸ پنج معیار اساسی را معرفی نمود:
\begin{enumerate}
    \item \textbf{میزان ایمنی سیستم:} برخی سیستم‌ها امنیت کامل دارند و برخی بر اساس پیچیدگی محاسباتی ایمن هستند.
    \item \textbf{اندازه کلید:} کلید باید کوچک و ساده باشد تا انتقال و حفظ آن آسان گردد.
    \item \textbf{پیچیدگی عملیات:} عملیات رمزگذاری و رمزگشایی باید ساده، سریع و کم‌هزینه باشد.
    \item \textbf{انتشار خطا (\lr{Error Propagation}):} خطا در انتقال نباید منجر به از دست رفتن کل پیام شود.
    \item \textbf{بسط یا گسترش پیام:} اندازه پیام نباید پس از رمزگذاری افزایش یابد.
\end{enumerate}

\subsection{ویژگی‌های نرم‌افزاری الگوریتم رمز}
برخی از ویژگی‌های کلیدی عبارتند از:
\begin{itemize}
    \item اجتناب از پرش‌های شرطی در حلقه‌های داخلی الگوریتم برای جلوگیری از اختلال در خط لوله پردازنده.
    \item استفاده محدود از عملگرهای سنگین مانند ضرب و تقسیم پیچیده.
    \item محدود کردن تعداد متغیرها برای استفاده بهینه از ثبات‌های پردازنده.
    \item کوچک نگه داشتن اندازه جداول (توصیه زیر ۴ کیلوبایت) برای تطابق با حافظه کش.
    \item بهره‌گیری از الگوهای موازی در عملگرها.
\end{itemize}

\section{نظریه‌ی آشوب و مفاهیم بنیادی آن}
\subsection{تعریف آشوب}
آشوب بیانگر نوعی رفتار غیرقابل پیش‌بینی در سامانه‌های پویاست که از نظمی درونی تبعیت می‌کند. پدیده‌ی آشوب معمولاً در سیستم‌های غیرخطی رخ می‌دهد که آینده‌ی رفتارشان به‌شدت به شرایط اولیه وابسته است.

\subsection{ویژگی‌های تئوری آشوب}
\begin{itemize}
    \item \textbf{الف) اثر پروانه‌ای (\lr{Butterfly Effect}):} تغییرات بسیار جزئی در شرایط اولیه‌ی معادلات می‌تواند منجر به تفاوت‌های بسیار بزرگ در نتایج نهایی شود.
    \item \textbf{ب) سازگاری پویا (\lr{Dynamic Adaptation}):} سیستم‌های آشوبناک دارای نوعی پویایی سازگار هستند و در تعامل با محیط تعادل برقرار می‌کنند.
    \item \textbf{پ) جذب‌کننده عجیب و پیچیده (\lr{Strange Attractor}):} مسیرهای زمانی سیستم در نهایت به سمت مجموعه‌ای پیچیده با ساختار فراکتالی میل می‌کنند.
    \item \textbf{ت) خودتشابهی (\lr{Self-Similarity}):} الگوی کلی رفتار سیستم در مقیاس‌های مختلف تکرار می‌شود؛ هر جزء کوچک تصویری از کل است.
    \item \textbf{ث) نمای لیاپانوف (\lr{Lyapunov Exponent}):} شاخصی برای تشخیص آشوب که میانگین نرخ واگرایی مسیرهای بسیار نزدیک را نشان می‌دهد. نمای مثبت نشانه آشوب است.
\end{itemize}

\section{ارتباط بین آشوب و رمزنگاری}
بین سیستم‌های رمزنگاری و آشوبناک رابطه‌ای بسیار نزدیک وجود دارد. شباهت‌های این دو سیستم عبارتند از:
\begin{itemize}
    \item \textbf{اغتشاش (\lr{Ergodicity}):} مشابه با مفهوم \lr{Confusion} در رمزنگاری.
    \item \textbf{حساسیت به شرایط اولیه:} مشابه با مفهوم \lr{Diffusion} و پخش خطا.
    \item \textbf{پارامترهای سیستم:} مشابه با کلید رمزنگاری.
\end{itemize}

\section{پیشینه پژوهش}
مطالعات انجام‌شده در زمینه رمزنگاری تصاویر رنگی با استفاده از سیستم‌های آشوبی و الگوریتم‌های بهینه‌سازی در دهه‌های اخیر به‌طور گسترده‌ای مورد توجه قرار گرفته است. در ادامه، به خلاصه‌ای از تحقیقات مرتبط اشاره می‌شود:

\begin{itemize}
    \item \textbf{ژانگ و لیو \cite{ref3}} در مطالعه خود ترکیب سیستم‌های آشوبی \lr{Henon Map} و \lr{Tent Map} را برای رمزنگاری تصاویر رنگی پیشنهاد دادند. نتایج این تحقیق نشان داد که این ترکیب، امنیت بیشتری نسبت به استفاده از یک سیستم آشوبی منفرد ارائه می‌دهد و الگوریتم پیشنهادی مقاومت خوبی در برابر حملات آماری دارد.
    \item \textbf{هو و ژو \cite{ref4}} یک مدل آشوبی نمایی (\lr{ECM}) برای تولید نگاشت‌های آشوبی یک‌بعدی با آشوب پایدار معرفی می‌کنند. \lr{ECM} با ترکیب دو سیستم آشوبی یک‌بعدی، نگاشت‌های جدیدی ایجاد می‌کند که در برابر تغییرات پارامتری مقاوم‌اند. نتایج ارزیابی عملکرد نشان می‌دهد که این نگاشت‌ها در فضای پارامتری گسترده‌ای آشوب پایدار دارند.
    \item \textbf{آلوارز و همکاران \cite{ref6}} به بررسی رمزنگاری تصاویر با استفاده از سیستم آشوبی لورنز می‌پردازند. در این پژوهش، ابتدا پیکسل‌های تصویر با استفاده از فرآیندهای جابه‌جایی و چرخش معکوس تخریب می‌شوند. نتایج حاصل نشان می‌دهند که میزان آنتروپی تصویر پس از رمزنگاری به \lr{7.9974} افزایش یافته و معیارهای \lr{NPCR} و \lr{UACI} به ترتیب \lr{99.65\%} و \lr{30.35\%} محاسبه شده‌اند.
    \item \textbf{خدادادی و زندوکیلی \cite{ref7}} روشی نوین برای رمزنگاری تصاویر رنگی با استفاده از ترکیب سیستم‌های آشوبی ارائه داده‌اند. الگوریتم پیشنهادی سه سری داده را با استفاده از سیستم آشوبی چن تولید کرده و با مقادیر \lr{RGB} تصویر ترکیب می‌کند. نوآوری اصلی در استفاده ترکیبی از دو سیستم آشوبی است که پیچیدگی فرآیند را افزایش می‌دهد.
    \item \textbf{وانگ و لی \cite{ref8}} نشان دادند که استفاده از سیستم آشوبی \lr{Logistic Map} در ترکیب با جایگشت و انتشار، می‌تواند یک الگوریتم رمزنگاری ایمن برای تصاویر ایجاد کند. با این حال، نیاز به بهبود کارایی محاسباتی همچنان یک چالش باقی می‌ماند.
    \item \textbf{مامت و همکاران \cite{ref9}} از یک سیستم رمزنگاری مبتنی بر آشوب برای رمزگذاری تصاویر رنگی استفاده کردند. در این روش ابتدا تصویر ورودی به کانال‌های قرمز، سبز و آبی تفکیک شده و هر کانال به‌صورت جداگانه پردازش می‌شود. کلیدهای رمزنگاری با استفاده از نقشه‌های آشوبی غیرخطی تولید می‌شوند که حساسیت بالایی به شرایط اولیه دارند.
    \item \textbf{لی و همکاران \cite{ref10}} با استفاده از سیستم آشوبی نمائی \lr{LEL}، دنباله‌های تصادفی با پیچیدگی بالا تولید کردند. نتایج پژوهش نشان داد که سیستم \lr{LEL} در مقایسه با سیستم‌های آشوبی رایج، قابلیت بالاتری در تولید توالی‌های غیرقابل پیش‌بینی دارد.
    \item \textbf{صادقیان \cite{ref11}} به بررسی روش جدیدی بر اساس ترکیب سیستم‌های آشوبی، الگوریتم ژنتیک و الگوریتم بهینه‌سازی ازدحام ذرات (\lr{PSO}) پرداخته است. ترکیب الگوریتم ژنتیک و \lr{PSO} به منظور بهینه‌سازی پارامترهای آشوبی و جلوگیری از جستجوی راکد به‌کار گرفته شده است.
    \item \textbf{گه و همکاران \cite{ref12}} یک الگوریتم رمزنگاری تصویر سریع و ایمن مبتنی بر فوق‌آشوب و عملیات برداری ارائه کردند. نتایج نشان می‌دهند که الگوریتم تمامی آزمون‌های \lr{NIST} را گذرانده و با زمان میانگین \lr{0.023} ثانیه، نیازهای انتقال محرمانه را برآورده می‌کند.
    \item \textbf{حسنی و همکاران \cite{ref13}} در سال ۲۰۲۴ یک روش نوین بر پایه ترکیب سه سیستم آشوبی \lr{Lorenz}، \lr{Henon} و \lr{2D-Logistic} طراحی کردند. این الگوریتم به‌صورت مستقل برای هر کانال رنگی عملیات جایگشت و انتشار انجام می‌دهد و مقاومت بالایی در برابر حملات آماری دارد.
\end{itemize}
