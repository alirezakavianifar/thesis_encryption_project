\chapter{مفاهیم پایه و پیشینه تحقیق}

\section{مقدمه}
از زمان انتشار مشاهده‌های ادوارد لورنز در زمینه‌ی رفتارهای آشوبناک، مفهوم «آشوب» به یکی از موضوعات مورد توجه در جامعه‌ی علمی تبدیل شد. نظریه‌ی آشوب شاخه‌ای از ریاضیات است که به تحلیل سیستم‌هایی می‌پردازد که رفتار آن‌ها به تغییرات کوچک در شرایط اولیه بسیار حساس است.

\section{رمزنگاری}
رمزنگاری فرآیندی است که طی آن اطلاعات به‌کمک یک الگوریتم خاص به صورتی غیرقابل‌فهم برای افراد غیرمجاز تبدیل می‌شود.

\subsection{تعریف رمزنگاری}
رمزنگاری فرآیندی است که طی آن اطلاعات به‌گونه‌ای دچار تغییر و بازآرایی می‌شوند که برای افراد غیرمجاز غیرقابل فهم باشد. هدف اصلی، حفظ محرمانگی و تمامیت اطلاعات است.

\subsection{شیوه‌های پایه رمزنگاری}
دو شیوه‌ی پایه وجود دارد: جابجایی (\lr{Transposition}) و جایگزینی (\lr{Substitution}). در جابجایی، موقعیت نمادها تغییر می‌کند و در جایگزینی، مقدار نمادها دگرگون می‌شود.

\subsection{انواع روش‌های رمزنگاری بر اساس نوع کلید}
الگوریتم‌های رمزنگاری به دو دسته متقارن (\lr{Symmetric}) و نامتقارن (\lr{Asymmetric}) تقسیم می‌شوند.

\subsection{مقایسه روش‌های رمزنگاری متقارن و نامتقارن}
در روش متقارن، کلید رمزگذاری و رمزگشایی یکسان است که منجر به سرعت بالاتر می‌شود اما مدیریت کلید دشوار است. در روش نامتقارن از جفت کلید عمومی و خصوصی استفاده می‌شود که امنیت توزیع کلید را افزایش می‌دهد اما سرعت کمتری دارد.

\subsection{تحلیل رمز: انواع حملات به سامانه‌های رمزنگاری}
حملات به سیستم‌های رمزنگاری شامل حملات متن آشکار منتخب، حملات فقط متن رمز، و حملات جستجوی فراگیر (\lr{Brute Force}) است. یک سیستم امن باید در برابر تمامی این حملات مقاوم باشد.

\subsection{امنیت سیستم‌های رمزنگاری}
امنیت یک سیستم بر اساس دو دیدگاه بررسی می‌شود:
\begin{itemize}
    \item \textbf{اصول ششگانه کرشهف:} امنیت سیستم نباید به محرمانه بودن الگوریتم وابسته باشد، بلکه باید تنها به محرمانه بودن کلید وابسته باشد.
    \item \textbf{معیارهای پنجگانه شانون:} شامل پخش (\lr{Diffusion}) و آشفتگی (\lr{Confusion}) است.
\end{itemize}

\section{رمزنگاری تصویر دیجیتال}
تصاویر دیجیتال به دلیل حجم بالا و همبستگی شدید پیکسل‌های مجاور، نیازمند الگوریتم‌های اختصاصی هستند.

\subsection{محدودیت‌های روش‌های سنتی}
روش‌هایی مانند \lr{AES} برای داده‌های متنی طراحی شده‌اند. اعمال مستقیم آن‌ها بر تصاویر به دلیل افزونگی بالای داده‌های تصویری، منجر به کارایی پایین و امنیت ناکافی در برابر تحلیل‌های آماری می‌شود.

\subsection{خصوصیات تصویر دیجیتال}
تصاویر دیجیتال از واحدهایی به نام پیکسل تشکیل شده‌اند. هر پیکسل در تصویر رنگی دارای سه مولفه \lr{RGB} است که هر کدام معمولاً ۸ بیت (بازه ۰-۲۵۵) فضا اشغال می‌کنند.

\section{نظریه‌ی آشوب و مفاهیم بنیادی آن}
نظریه‌ی آشوب به مطالعه سیستم‌های دینامیکی غیرخطی می‌پردازد که علی‌رغم ماهیت قطعی، رفتاری شبه‌تصادفی دارند.

\subsection{ویژگی‌های تئوری آشوب}
\begin{itemize}
    \item \textbf{حساسیت به شرایط اولیه:} تغییرات بسیار جزئی در ورودی منجر به خروجی‌های کاملاً متفاوت می‌شود (اثر پروانه‌ای).
    \item \textbf{غیرخطی بودن:} روابط سیستم به‌صورت خطی قابل بیان نیستند.
    \item \textbf{رفتار شبه‌تصادفی:} خروجی‌ها شبیه نویز هستند اما دارای نظم درونی می‌باشند.
\end{itemize}

\subsection{نمونه‌هایی از سیستم‌های آشوبی}
از جمله سیستم‌های مشهور می‌توان به نگاشت لجستیک، سیستم لورنز و سیستم \lr{Chen} اشاره کرد. سیستم \lr{Chen} به دلیل پیچیدگی بالاتر نسبت به لورنز، برای رمزنگاری ترجیح داده می‌شود.

\section{ارتباط بین آشوب و رمزنگاری}
شباهت‌های ساختاری میان آشوب و رمزنگاری (اغتشاش/سردرگمی و حساسیت/انتشار) استفاده از این سیستم‌ها را توجیه می‌کند.

\section{پیشینه پژوهش}
در سال‌های اخیر، استفاده از نقشه‌های آشوبی نمایی (\lr{ECM}) به دلیل پایداری بیشتر و عدم تخریب ویژگی‌های آشوبی مورد توجه قرار گرفته است. پژوهش‌های ژانگ (۲۰۲۳) و هو (۲۰۱۹) مبانی این رویکرد را بنا نهاده‌اند.
