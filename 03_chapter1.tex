\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\item
  \protect\hypertarget{_Toc217957694}{}{}\RL{بيان مسأله}
\end{enumerate}

\RL{در عصر دیجیتال، تبادل اطلاعات به‌ویژه در قالب تصاویر، به بخشی جدایی‌ناپذیر از ارتباطات، تجارت، سیستم‌های نظارتی، و رسانه‌های اجتماعی تبدیل شده است. در این میان، تصاویر دیجیتال به دلیل نقش کلیدی در انتقال داده‌های بصری، به یکی از مهم‌ترین انواع اطلاعات چندرسانه‌ای بدل شده‌اند که حجم بالایی از داده را در خود جای می‌دهند و کاربرد گسترده‌ای در شبکه‌های ارتباطی، سامانه‌های ذخیره‌سازی و تحلیل داده دارند {[}1{]}.}

\RL{با گسترش دسترسی به فضای مجازی و تهدیدهای فزاینده در حوزه امنیت سایبری، حفاظت از حریم خصوصی اطلاعات تصویری به یکی از دغدغه‌های اصلی پژوهشگران و توسعه‌دهندگان امنیت اطلاعات تبدیل شده است. برخلاف داده‌های متنی که رمزنگاری آن‌ها با الگوریتم‌های استانداردی نظیر AES و RSA به‌خوبی قابل انجام است، تصاویر دیجیتال به دلیل ویژگی‌هایی همچون حجم زیاد، ساختار دوبعدی، همبستگی آماری بالا میان پیکسل‌های مجاور، و حساسیت بالا به تغییرات، به الگوریتم‌های رمزنگاری تخصصی‌تر و پیچیده‌تری نیاز دارند {[}2و3{]}.}

\RL{مهم‌ترین چالش‌ها در رمزنگاری تصاویر دیجیتال را می‌توان به شرح زیر برشمرد: حفظ ویژگی‌های آماری تصویر پس از رمزنگاری؛ ایجاد مقاومت در برابر انواع حملات نظیر تحلیل آماری، حملات تمایزی و جستجوی فراگیر؛ کاهش بار محاسباتی جهت کاربرد در سامانه‌های بلادرنگ؛ پیشگیری از تخریب داده‌های تصویری طی فرآیند رمزنگاری و رمزگشایی.}

\item
  \protect\hypertarget{_Toc217957699}{}{}\RL{مفاهیم کلیدی تحقیق}
\end{enumerate}

\textbf{\RL{رمزنگاری (Encryption):}} فرایندی ریاضیاتی است که طی آن، داده‌های اصلی تبدیل می‌شوند.

\textbf{\RL{سیستم آشوبی (Chaotic System):}} سیستم‌های دینامیکی غیرخطی هستند که رفتار آن‌ها به شدت به شرایط اولیه وابسته است.

\textbf{\RL{جایگشت (Permutation):}} یکی از مراحل اصلی در رمزنگاری تصویر است.

\RL{در این پایان‌نامه، یک چارچوب نوین برای رمزنگاری تصاویر رنگی مبتنی بر سیستم‌های آشوبی نمایی ارائه شده است. مهم‌ترین سهم‌های علمی این پژوهش شامل: ارائه الگوریتم رمزنگاری مبتنی بر مدل آشوبی نمایی، استفاده هم‌زمان از چند سیستم آشوبی نمایی (۹ نگاشت متفاوت)، طراحی سازوکار انتشار اصلاح‌شده و معکوس‌پذیر، بهره‌گیری از سیستم آشوبی پیوسته Chen برای تولید جریان کلید، و انجام مجموعه‌ای جامع از تحلیل‌های امنیتی است.}
