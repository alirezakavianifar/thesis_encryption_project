\chapter{معرفی الگوریتم پیشنهادی}

\section{مقدمه}
در این فصل، به تشریح دقیق الگوریتم پیشنهادی رمزنگاری تصاویر رنگی و نتایج تجربی آن پرداخته می‌شود. هدف این فصل، ارائه یک تصویر کامل از معماری الگوریتم، فرآیندهای رمزنگاری و تحلیل‌های امنیتی و آماری است. نتایج ارائه‌شده شامل تحلیل بصری تصاویر، هیستوگرام‌ها، آنتروپی، کاهش همبستگی پیکسل‌ها، حساسیت به کلید و مقاومت در برابر حملات رمزنگاری می‌باشد. تمامی شبیه‌سازی‌ها با استفاده از نرم‌افزار \lr{MATLAB} و تحلیل‌های عددی در محیط \lr{Python} انجام شده است. این فصل به‌عنوان هسته تجربی پژوهش، مجموعه‌ای از تحلیل‌های چندلایه و ارزیابی‌های کمی و کیفی را ارائه می‌کند تا میزان اثربخشی الگوریتم رمزنگاری پیشنهادی با اتکا به شواهد عینی و نتایج عددی مورد سنجش قرار گیرد.

ساختار فصل بر پایه سه ستون اصلی بنا شده است:
\begin{enumerate}
    \item معرفی الگوریتم پیشنهادی
    \item ارائه خروجی‌های بصری برای سنجش کیفی رفتار الگوریتم
    \item تحلیل دقیق شاخص‌های آماری بنیادین رمزنگاری تصویر
    \item ارزیابی امنیتی-مقاومتی در برابر حملات آماری، دیفرانسیلی و ساختاری
\end{enumerate}

هدف از این فصل، نه تنها تأیید عملی فرضیه‌های تحقیق، بلکه نشان دادن برتری نسبی روش پیشنهادی در مقایسه با معماری‌های مشابه مبتنی بر سیستم‌های آشوبی است. بنابراین، هر بخش از تحلیل‌ها به‌صورت مستقیم به یک یا چند فرضیه پژوهش متصل شده و نشان می‌دهد چگونه اجزای الگوریتم از سطح نظری به عملی تبدیل شده‌اند.

\section{انتخاب سیستم‌های آشوبی و مجموعه داده}
\subsection{انتخاب سیستم‌های آشوبی}
به جای استفاده از یک سیستم آشوبی منفرد، از ۹ سیستم نمایی بالا برای رمزنگاری تصاویر استفاده شده است. این انتخاب باعث افزایش تصادفی بودن و پیچیدگی فرآیند رمزنگاری شده و امنیت الگوریتم را به طور قابل توجهی بهبود می‌بخشد.

\subsection{انتخاب مجموعه داده}
برای ارزیابی عملکرد الگوریتم، مجموعه داده استاندارد \lr{USC-SIPI} مورد استفاده قرار گرفته است. این مجموعه شامل تصاویر رنگی با ابعاد ۲۵۶×۲۵۶ و ۵۱۲×۵۱۲ و ویژگی‌های متنوع می‌باشد.

\section{فرآیند رمزنگاری}
فرآیند رمزنگاری تصاویر در پنج مرحله اصلی انجام می‌شود:

\begin{enumerate}
    \item \textbf{تولید کلیدهای اولیه:} مقادیر اولیه برای سیستم‌های آشوبی با استفاده از اعداد تصادفی انتخاب می‌شوند. این مقادیر نقش کلید رمزنگاری را ایفا می‌کنند و امنیت الگوریتم را تضمین می‌کنند.
    \item \textbf{جایگشت پیکسل‌ها با سیستم \lr{Chen}:} برای شکستن همبستگی میان پیکسل‌ها، از سیستم آشوبی چن برای تولید دنباله‌ای تصادفی استفاده می‌شود. جایگشت به گونه‌ای انجام می‌شود که موقعیت پیکسل‌ها در تصویر تغییر کند:
    \begin{equation}
    \begin{cases}
        \frac{dx}{dt} = a(y - x) \\
        \frac{dy}{dt} = (c - a)x - xz + cy \\
        \frac{dz}{dt} = xy - bz
    \end{cases}
    \end{equation}
    پارامترهای $a$، $b$ و $c$ مقادیر ثابت هستند.
    \item \textbf{رمزنگاری با سیستم‌های آشوبی نمایی:} برای هر پیکسل، یک عدد بین ۱ تا ۹ تولید می‌شود که مشخص می‌کند کدام یک از ۹ سیستم آشوبی نمایی برای رمزنگاری آن پیکسل به کار رود.
    \item \textbf{انتشار مقادیر رنگی:} پس از جایگشت، مقادیر رنگی پیکسل‌ها با مقادیر تولید شده از سیستم آشوبی مربوطه ترکیب می‌شوند تا پیچیدگی و تصادفی بودن بیشتر شود.
    \item \textbf{ترکیب نهایی:} خروجی مرحله قبلی با یکی دیگر از سیستم‌های آشوبی ترکیب شده و فرآیند رمزنگاری به‌صورت نهایی تکمیل می‌شود.
\end{enumerate}

\section{پیاده‌سازی و تحلیل}
\begin{itemize}
    \item \textbf{پیاده‌سازی الگوریتم:} برای شبیه‌سازی الگوریتم از محیط \lr{Python} با استفاده از کتابخانه‌های علمی \lr{SciPy}، \lr{OpenCV}، و \lr{NumPy} انجام می‌شود.
    \item \textbf{ارزیابی امنیتی:} معیارهایی مانند آنتروپی شانون، تحلیل همبستگی، حساسیت به شرایط اولیه و مقاومت در برابر حملات رمزنگاری مورد بررسی قرار می‌گیرند.
    \item \textbf{تحلیل زمان اجرا:} زمان لازم برای رمزنگاری و رمزگشایی اندازه‌گیری شده و اثر اندازه تصویر بر زمان اجرا بررسی می‌شود.
\end{itemize}

\section{معماری نهایی و عملیاتی الگوریتم پیشنهادی}
الگوریتم پیشنهادی بر رویکرد چندآشوبیِ ترکیبی و نمایی استوار است که موجب جهش قابل توجه در پیچیدگی محاسباتی-ساختاری می‌شود.

\subsection{تحلیل ساختاری مراحل پیاده‌سازی}

\subsubsection{۱. تولید مسیرهای آشوبی مبتنی بر سیستم سه‌بعدی \lr{Chen}}
حل عددی این سیستم با \lr{solve\_ivp} نه‌تنها پایداری محاسباتی را تضمین می‌کند بلکه به‌عنوان یک مولد با کیفیت برای ایجاد دنباله‌های شبه‌تصادفی در مرحله جایگشت عمل می‌کند.

\subsubsection{۲. تولید بردار جایگشت از دنباله آشوبی}
پس از نرمال‌سازی خروجی‌ها، بردارهای جایگشت‌های یک‌به‌یک تولید می‌شود که موجب تخریب کامل ساختار هندسی تصویر می‌شود. این بردارها فاقد هرگونه الگوی خطی یا دوره‌ای هستند.

\subsubsection{۳. تعریف ۹ سیستم آشوبی نمایی ترکیبی}
این ۹ سیستم با ساختار نمایی-تکراری، توان محاسبات غیرخطی بسیار بالایی دارند و هرکدام به‌صورت مستقل دنباله‌های بایت پیچیده‌ای تولید می‌کنند.

\subsubsection{۴. انتخاب تصادفی تابع نمایی برای هر پیکسل}
برای هر پیکسل، سیستم رمزکننده یکی از ۹ نقشه آشوبی را بر اساس انتخاب‌گر تصادفی مشخص می‌کند. این یک ساختار فوق‌العاده پیچیده در سطح انتشار ایجاد می‌کند.

\subsubsection{۵. فاز انتشار مبتنی بر \lr{XOR}}
دنباله‌های بایت‌های تولیدشده به‌صورت مستقیم در عملیات \lr{XOR} با پیکسل‌های جایگشت‌شده به‌کار گرفته می‌شوند. این مرحله عامل اصلی مقاومت در برابر حملات تفاضلی است.

\subsubsection{۶. رمزگشایی معکوس}
از آنجا که \lr{XOR} یک عملیات برگشت‌پذیر است و جایگشت نیز معکوس‌پذیر طراحی شده، الگوریتم رمزگشایی می‌تواند تصویر اصلی را بدون هیچ‌گونه خطا بازیابی کند (\lr{Lossless recovery}). تصویر بازیابی‌شده به‌صورت صددرصد با تصویر اصلی منطبق بوده و معکوس‌پذیری کامل الگوریتم تضمین می‌شود.
