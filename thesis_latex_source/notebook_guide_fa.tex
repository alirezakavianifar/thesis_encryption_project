\section{فاز ۱: سیستم آشوبی چن و نگاشت‌های یک‌بعدی پایه}
سیستم آشوبی چن یک سیستم پیوسته سه‌بعدی است که رفتار آشوبی بسیار پیچیده و جاذبه‌های آشوبی چندگانه‌ای از خود نشان می‌دهد. معادلات دیفرانسیل این سیستم به صورت زیر تعریف می‌شوند:
\begin{equation}
\label{eq:chen}
\begin{aligned}
\frac{dx}{dt} &= a(y - x) \\
\frac{dy}{dt} &= (c - a)x - xz + cy \\
\frac{dz}{dt} &= xy - bz
\end{aligned}
\end{equation}
که در آن پارامترهای سیستم معمولاً در مقادیر استاندارد آشوبی زیر تنظیم می‌شوند:
\begin{equation}
a = 35, \quad b = 3, \quad c = 28
\end{equation}

\subsection{تحلیل تابع حل عددی و حذف فاز گذرا (\lr{Warm-up})}
در تابع \texttt{generate\_chen\_sequence}، سیستم چن با استفاده از تابع \texttt{solve\_ivp} در بازه زمانی با گام‌های زمانی $dt = 0.002$ حل می‌شود. یکی از نکات بسیار حیاتی در تدریس به دانشجو، متغیر \texttt{warmup=1000} است. 
به دانشجو توضیح دهید که سیستم‌های آشوبی در ابتدای حرکت خود از شرایط اولیه، یک فاز گذرا (\lr{Transient Phase}) را طی می‌کنند که در آن رفتار سیستم هنوز به پایداری آشوبی و توزیع یکنواخت نرسیده است. با نادیده گرفتن ۱۰۰۰ گام اول سیستم، دنباله‌ای کاملاً تصادفی و نامشخص از نظر آماری برای تولید کلید و جایگشت به دست می‌آوریم که امنیت را به شدت افزایش می‌دهد.

\subsection{نگاشت‌های پایه یک‌بعدی}
سه نگاشت آشوبی معروف به عنوان توابع پایه تعریف شده‌اند:
\begin{enumerate}
    \item \textbf{نگاشت لجستیک (\lr{Logistic Map})}:
    \begin{equation}
    x_{n+1} = r x_n (1 - x_n)
    \end{equation}
    در $r=4.0$ این نگاشت دارای آشوب کامل در بازه $[0, 1]$ است.
    \item \textbf{نگاشت سینوسی (\lr{Sine Map})}:
    \begin{equation}
    x_{n+1} = \frac{r}{4} \sin(\pi x_n)
    \end{equation}
    در $r=1.0$ رفتار آشوبی پایداری نشان می‌دهد.
    \item \textbf{نگاشت خیمه‌ای (\lr{Tent Map})}:
    \begin{equation}
    x_{n+1} = \begin{cases} 
      2 x_n & x_n < 0.5 \\
      2(1 - x_n) & x_n \ge 0.5 
   \end{cases}
    \end{equation}
\end{enumerate}

\section{فاز ۲: خانواده نه نگاشت آشوبی نمایی (\lr{ECM})}
برای غلبه بر مشکلات نگاشت‌های یک‌بعدی پایه (مانند پنجره‌های تناوبی ناخواسته و توزیع غیریکنواخت)، ساختار جدیدی تحت عنوان نگاشت آشوبی نمایی (\lr{Exponential Chaotic Map - ECM}) معرفی شده است. این ساختار از ترکیب دو نگاشت پایه (یکی داخلی و دیگری خارجی) همراه با یک ضرب‌کننده نمایی تشکیل می‌شود. 

\subsection{معادله ساختاری \lr{ECM}}
اگر نگاشت داخلی را $F_{\mathrm{in}}$ و نگاشت خارجی را $F_{\mathrm{out}}$ بنامیم، یک گام از نگاشت نمایی به صورت زیر محاسبه می‌شود:
\begin{equation}
x_{n+1} = F_{\mathrm{out}} \left( e^{\mu \cdot F_{\mathrm{in}}(x_n)} \pmod 1 \right)
\end{equation}
که در آن $\mu = 3.8$ پارامتر کنترل‌کننده نمایی است. این ساختار تضمین می‌کند که خروجی نگاشت داخلی پس از قرارگیری در نمای عدد نپر و اعمال باقی‌مانده بر یک، حساسیت فوق‌العاده بالایی نسبت به شرایط اولیه پیدا کند.

\subsection{نه ترکیب مختلف}
با داشتن ۳ نگاشت پایه، می‌توانیم $3 \times 3 = 9$ سیستم آشوبی نمایی مستقل بسازیم:

\section{فاز ۳: انتخاب‌گر آشوبی پویا و لایه \lr{XOR} نهایی}
اگر کل تصویر را تنها با یک نگاشت آشوبی ثابت رمز کنیم، الگوی آماری سیستم آشوبی ممکن است در هیستوگرام یا آنالیزهای تفاضلی لو برود. برای حل این مشکل، دو ایده در فاز ۳ پیاده‌سازی شده است:

\subsection{انتخاب‌گر آشوبی (\lr{Selector Sequence})}
با استفاده از یک نگاشت لجستیک مستقل با شرایط اولیه $x_{\mathrm{sel}}$، به ازای هر پیکسل تصویر، یک عدد صحیح بین ۱ تا ۹ تولید می‌کنیم:
\begin{equation}
\mathrm{index}_k = \lfloor x_k \times 9 \rfloor + 1
\end{equation}
این شاخص مشخص می‌کند که برای رمزگذاری پیکسل $k$-ام، خروجی کدام یک از ۹ نگاشت ECM انتخاب شود. این فرآیند باعث سوئیچینگ داینامیک (\lr{Dynamic Switching}) بین فضای ۹ سیستم آشوبی می‌شود که تحلیل‌های آماری هکرها را کاملاً خنثی می‌کند.

\subsection{دنباله لایه \lr{XOR} نهایی}
به منظور پوشش‌دهی کامل و از بین بردن هرگونه اثر آماری باقی‌مانده از لایه‌های قبلی، یک دنباله لایه XOR نهایی مستقل با استفاده از شرایط اولیه $x_{\mathrm{final}}$ تولید می‌شود. این دنباله مقادیر اعشاری آشوبی را به بایت‌های ۸ بیتی مپ می‌کند:
\begin{equation}
\mathrm{KeyFinal}_k = \lfloor x_k \times 256 \rfloor \pmod{256}
\end{equation}

\section{فاز ۴: فرآیند رمزنگاری، رمزگشایی و یکنواخت‌سازی کامل هیستوگرام}
این فاز هسته اصلی الگوریتم را تشکیل می‌دهد. رمزنگاری تصویر رنگی شامل مراحل زیر است که روی هر سه کانال رنگی (قرمز، سبز، آبی) اعمال می‌شود:

\subsection{مراحل فرآیند رمزنگاری}
\begin{enumerate}
    \item \textbf{جایگشت پیکسل‌ها با سیستم چن (\lr{Permutation})}:
    ابتدا دنباله آشوبی چن به طول $N = H \times W$ تولید می‌شود. مقادیر این دنباله سورت شده و اندیس‌های سورت‌شده، بردار جایگشت (\lr{Permutation Vector}) را می‌سازند. پیکسل‌های هر سه کانال تصویر رنگی بر اساس این بردار جابه‌جا می‌شوند تا وابستگی مکانی شدید پیکسل‌های مجاور از بین برود.
    \item \textbf{انتشار با} \lr{\textbf{XOR}} \textbf{انتخابی} \lr{\textbf{ECM}} \textbf{(\lr{Diffusion})}:
    برای هر پیکسل، شاخص انتخاب‌گر سیستم آشوبی محاسبه شده و مقدار آن در سیستم \lr{ECM} مربوطه یک گام به جلو حرکت داده می‌شود. مقدار حاصل از این سیستم با بایت پیکسل جایگشت‌یافته \lr{XOR} می‌شود.
    \item \textbf{لایه} \lr{\textbf{XOR}} \textbf{نهایی}:
    تصویر حاصل با دنباله تصادفی نهایی \lr{XOR} می‌شود تا توزیع پیکسل‌ها تصادفی‌تر گردد.
    \item \textbf{یکنواخت‌سازی هیستوگرام (\lr{Histogram Balancing})}:
    این مرحله بسیار نوآورانه است. حتی پس از رمزنگاری آشوبی، هیستوگرام تصویر کاملاً تخت و یکنواخت نیست. برای رسیدن به یکنواختی کامل (انحراف معیار صفر در هیستوگرام)، تابع \texttt{balance\_histogram\_flat} مقادیر پیکسل‌های اضافی را شناسایی کرده و آن‌ها را با مقادیری که با کمبود مواجه هستند تعویض می‌کند. ترتیب تعویض بر اساس مقادیر اعشاری یک دنباله آشوبی مستقل دیگر انجام می‌شود. برای حفظ قابلیت رمزگشایی کامل (\lr{Lossless Decryption}), مختصات پیکسل‌های جابه‌جاشده به همراه مقدار اولیه آن‌ها در قالب یک لیست به عنوان فراداده (\texttt{swaps}) ذخیره شده و به همراه تصویر ارسال می‌شود.
\end{enumerate}

\subsection{مراحل فرآیند رمزگشایی}
رمزگشایی دقیقاً معکوس فرآیند رمزنگاری است:
\begin{enumerate}
    \item ابتدا با استفاده از اطلاعات \texttt{swaps} ذخیره‌شده در متادیتا، جابه‌جایی‌های یکنواخت‌سازی هیستوگرام به صورت معکوس (از انتها به ابتدا) برگردانده می‌شوند تا تصویر رمز شده اولیه بازسازی شود.
    \item لایه \lr{XOR} نهایی با همان شرایط اولیه بازسازی و معکوس می‌شود.
    \item فرآیند \lr{XOR} انتخابی \lr{ECM} به صورت معکوس اعمال می‌شود.
    \item بردار جایگشت معکوس (\lr{Inverse Permutation Vector}) روی تصویر اعمال شده و تصویر اصلی بدون کوچکترین افت کیفیت بازسازی می‌گردد.
\end{enumerate}

\section{فاز ۵: معیارهای ارزیابی امنیتی}
برای اثبات امنیت الگوریتم، محاسبات آماری زیر روی تصاویر انجام می‌شود:

\subsection{آنتروپی شانون (\lr{Shannon Entropy})}
میزان تصادفی بودن اطلاعات تصویر را نشان می‌دهد:
\begin{equation}
H(s) = -\sum_{i=0}^{255} p(s_i) \log_2 p(s_i)
\end{equation}
برای یک تصویر رمزشده ۸ بیتی ایده‌آل، آنتروپی باید دقیقاً برابر $8.0$ باشد. به علت وجود مرحله یکنواخت‌سازی هیستوگرام، آنتروپی تصاویر رمزشده در این پروژه دقیقاً برابر $8.0000$ به دست می‌آید.

\subsection{همبستگی پیکسل‌های مجاور (\lr{Pixel Correlation})}
در تصاویر معمولی همبستگی پیکسل‌های مجاور بسیار نزدیک به ۱ است. در یک تصویر رمزشده مناسب، این همبستگی باید به صفر نزدیک شود. این مقدار در سه جهت افقی (\lr{H})، عمودی (\lr{V}) و قطری (\lr{D}) محاسبه می‌شود.

\subsection{حساسیت به کلید و حملات تفاضلی (\lr{NPCR} و \lr{UACI})}
\begin{itemize}
    \item \textbf{\lr{\textbf{NPCR}} (تعداد پیکسل‌های تغییر یافته)}: میزان تغییر در تصویر رمزشده در اثر تغییر یک بیت در کلید اصلی یا تصویر اصلی. مقدار ایده‌آل آن $>99.6\%$ است.
    \item \textbf{\lr{\textbf{UACI}} (میانگین شدت تغییرات)}: میانگین تغییرات شدت روشنایی بین دو تصویر رمزشده. مقدار ایده‌آل آن نزدیک به $33.46\%$ است.
\end{itemize}
در شبیه‌سازی، شرط اولیه $x_0$ به اندازه ناچیز $10^{-15}$ تغییر داده شده و حساسیت فوق‌العاده بالای الگوریتم اثبات می‌شود.

\subsection{معیار ارزیابی خطای بازسازی (\lr{MSE} و \lr{PSNR})}
مقادیر \lr{MSE} تفاضل پیکسلی تصویر اصلی و بازیابی‌شده را نشان می‌دهد که باید دقیقاً $0.0$ باشد و \lr{PSNR} نیز باید برابر بی‌نهایت ($\infty$) به دست آید که نشان‌دهنده بدون اتلاف بودن کامل فرآیند است.

\chapter{بخش چهارم: تحلیل و تفسیر نتایج نهایی آزمایش‌ها}

برای بخش پایانی تدریس، خلاصه نتایج عددی زیر را به دانشجو ارائه داده و تحلیل کنید:

\section{تحلیل آنتروپی شانون}
\lr{Shannon Entropy} تصاویر رمزشده دقیقاً $8.0$ است. این یک نتیجه ایده‌آل در امنیت اطلاعات است زیرا نشان می‌دهد هیچ‌گونه نشت اطلاعات آماری وجود ندارد و هکر با آنالیز توزیع فراوانی بایت‌ها نمی‌تواند هیچ اطلاعاتی از تصویر اصلی به دست آورد.

\section{تحلیل نمودار پراکندگی و همبستگی پیکسل‌ها}
نمودار پراکندگی همبستگی نشان می‌دهد که وابستگی پیکسلی در تصویر رمزشده کاملاً از بین رفته است. این امر نشان‌دهنده شکست کامل حملات آماری و بازیابی الگوهای تصویر است.

\section{مقاومت در برابر حملات تفاضلی (تست کلید)}
تغییر جزئی کلید اولیه به میزان $10^{-15}$ منجر به تولید تصویر رمزشده کاملاً متفاوتی می‌شود که مقادیر \lr{NPCR} نزدیک به $99\%$ و \lr{UACI} نزدیک به $33\%$ را تأیید می‌کند. این نشان می‌دهد الگوریتم در برابر حملات تفاضلی کاملاً مقاوم است.

\section{بازیابی بدون اتلاف (\lr{Lossless})}
با توجه به اینکه مقدار تفاضل میانگین مربعات خطا (\lr{MSE}) صفر و نسبت سیگنال به نویز (\lr{PSNR}) برابر بی‌نهایت است، اثبات می‌شود که الگوریتم علی‌رغم تغییرات هیستوگرام، کاملاً برگشت‌پذیر و بدون اتلاف است و برای کاربردهای حساس نظامی و پزشکی مناسب می‌باشد.
